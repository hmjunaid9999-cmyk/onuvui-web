rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return signedIn()
        && request.auth.token.email == "hmjunaid9999@gmail.com";
    }

    function isOwner(uid) {
      return signedIn() && request.auth.uid == uid;
    }

    // ---------- Existing site settings ----------
    match /settings/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // ---------- Existing posts ----------
    match /posts/{postId} {
      allow read: if true;
      allow create: if signedIn();
      allow update: if signedIn()
        && (isAdmin() || resource.data.authorUid == request.auth.uid);
      allow delete: if signedIn()
        && (isAdmin() || resource.data.authorUid == request.auth.uid);

      // ---------- Comments নেস্টেড হিসেবে posts এর ভেতরে (আসল কোড এভাবেই লেখে) ----------
      match /comments/{commentId} {
        allow read: if true;
        allow create: if signedIn()
          && request.resource.data.uid == request.auth.uid;
        allow update: if signedIn()
          && (isAdmin() || resource.data.uid == request.auth.uid);
        allow delete: if signedIn()
          && (isAdmin() || resource.data.uid == request.auth.uid);
      }
    }

    // ---------- Old top-level comments (unused by current app code, kept harmless) ----------
    match /comments/{commentId} {
      allow read: if true;
      allow create: if signedIn();
      allow update: if signedIn()
        && (isAdmin() || resource.data.uid == request.auth.uid);
      allow delete: if signedIn()
        && (isAdmin() || resource.data.uid == request.auth.uid);
    }

    // ---------- Users / Point balance ----------
    match /users/{userId} {
      allow read: if signedIn() && (request.auth.uid == userId || isAdmin());

      // New users can create their own profile.
      // (আগে points==0 শর্ত ছিল, যেটা video.html-এর লগইন কোড points ফিল্ড না পাঠানোয়
      //  নতুন ইউজারদের প্রোফাইল তৈরি ব্লক করে দিচ্ছিল — তাই শর্তটা সরানো হলো)
      allow create: if isOwner(userId);

      // Admin can change anything.
      allow update: if isAdmin();

      // User can update only non-financial profile fields.
      // Point/balance fields are protected from client-side changes.
      allow update: if isOwner(userId)
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly([
            'displayName',
            'photoURL',
            'name',
            'updatedAt'
          ]);

      allow delete: if isAdmin();

      // ---------- Point history ----------
      match /pointTransactions/{transactionId} {
        allow read: if isOwner(userId) || isAdmin();
        allow create: if isAdmin();
        allow update, delete: if isAdmin();
      }
    }

    // ---------- Videos ----------
    match /videos/{videoId} {
      allow read: if true;

      // A user may create only a video belonging to themselves.
      // The app should send ownerUid and pointsSpent.
      allow create: if signedIn()
        && request.resource.data.ownerUid == request.auth.uid
        && request.resource.data.pointsSpent is int
        && request.resource.data.pointsSpent >= 0;

      // Users cannot edit an existing video. Admin can.
      allow update: if isAdmin();

      // Owner can delete own video; admin can delete any.
      allow delete: if signedIn()
        && (isAdmin() || resource.data.ownerUid == request.auth.uid);

      // ---------- Video statistics ----------
      match /stats/{statsId} {
        allow read: if true;

        // Stats are created/updated by signed-in users.
        // The client must use increment() for view/like counts.
        allow create, update: if signedIn()
          && statsId == "counts";

        allow delete: if isAdmin();
      }

      // ---------- Likes ----------
      match /likes/{userId} {
        allow read: if true;
        allow create: if isOwner(userId);
        allow delete: if isOwner(userId);
        allow update: if false;
      }

      // ---------- Video comments ----------
      match /comments/{commentId} {
        allow read: if true;
        allow create: if signedIn()
          && request.resource.data.uid == request.auth.uid;
        allow update: if signedIn()
          && (isAdmin() || resource.data.uid == request.auth.uid);
        allow delete: if signedIn()
          && (isAdmin() || resource.data.uid == request.auth.uid);
      }
    }

    // ---------- Advertisements ----------
    match /advertisements/{adId} {
      allow read: if true;

      allow create: if signedIn()
        && request.resource.data.ownerUid == request.auth.uid;

      // User-created ads are not editable after submission.
      allow update: if isAdmin();

      allow delete: if signedIn()
        && (isAdmin() || resource.data.ownerUid == request.auth.uid);

      match /stats/{statsId} {
        allow read: if true;
        allow create, update: if signedIn() && statsId == "counts";
        allow delete: if isAdmin();
      }
    }

    // ---------- Payment requests ----------
    match /paymentRequests/{requestId} {
      allow read: if signedIn()
        && (isAdmin() || resource.data.uid == request.auth.uid);

      allow create: if signedIn()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.status == "pending";

      // Only admin can approve/reject or otherwise modify a request.
      allow update, delete: if isAdmin();
    }

    // ---------- Point purchase packages/settings ----------
    match /pointPackages/{packageId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // ---------- Deny everything not explicitly listed ----------
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
